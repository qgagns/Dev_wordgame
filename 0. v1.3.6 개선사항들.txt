(() => {
    // 1. 일반 영지 단어들 중 총 20마리 추출
    const allNormalWords = [];
    worldMaps.forEach(m => {
        const idLower = (m.id || "").toLowerCase();
        if (idLower.includes('net_') || idLower.includes('irreg')) return;
        m.words.forEach(w => allNormalWords.push(w));
    });

    const mist10 = allNormalWords.slice(0, 10);   // 1~10번: 침식 정령 (구출용)
    const hurt10 = allNormalWords.slice(10, 20);  // 11~20번: 상처 입은 정령 (성소용)

    const mistKeys = mist10.map(w => w.eng);
    const hurtKeys = hurt10.map(w => w.eng);

    const now = Date.now();

    // 2. 다른 모든 단어는 침식되지 않도록 안전 면역(3단계) 처리
    allNormalWords.forEach(w => {
        if (!mistKeys.includes(w.eng)) {
            wordMasteryTimestamps[w.eng] = now;
            wordDefenseCounts[w.eng] = 3; // 면역
        }
    });

    // 3. [침식 정령 10마리 주입] 계약 완료(3점) + 시간 만료
    mist10.forEach(w => {
        wordMastery[w.eng] = 3;
        wordMasteryTimestamps[w.eng] = 0; // 쿨타임 만료
        wordDefenseCounts[w.eng] = 0;     // 침식 발생
    });

    // 4. [상처 입은 정령 10마리 주입] 누적 오답 2회 ➔ 치유실 즉시 면담 대기
    hurt10.forEach(w => {
        wordCounselingData[w.eng] = {
            wrongAccum: 2,
            step: 0, // 1차 즉시 면담
            lastTime: now
        };
    });

    // 5. 오늘치 구출 완료 플래그 초기화
    lastEmergencyClearedDate = "";
    safeStorageSet('spirit_emergency_cleared_date', "");
    const todayStr = getTodayKSTString();
    safeStorageSet(`spirit_defense_today_count_${todayStr}`, 0);

    // 6. 데이터 영구 저장 및 화면 동기화
    if (worldMaps[0]) safeStorageSet(`spirit_mastery_${worldMaps[0].id}`, wordMastery);
    safeStorageSet('spirit_mastery_timestamps', wordMasteryTimestamps);
    safeStorageSet('spirit_defense_counts', wordDefenseCounts);
    safeStorageSet('spirit_counseling_data', wordCounselingData);
    renderTitleScreen();

    console.log("✨ [생성 완료] 🚨 침식 정령 10마리 + 🍂 상처 입은 정령 10마리가 배치되었습니다!");
})();

[코드 응답 규칙]
1. 사용자가 명시적으로 "전체 코드"를 요구하지 않는 한, 전체 파일을 다시 출력하지 않는다.
2. 수정이 필요한 특정 함수, 객체, 또는 diff 형식 블록만 제공한다.
3. 코드가 삽입되거나 수정되어야 할 위치(함수명 또는 주변 코드 맥락)를 명확히 안내한다.
4. 설명은 장황하지 않게 핵심 변경 이유 위주로 간결하게 작성한다.
5. 코드를 작성할 때는 CSS를 적극 활용하여 지속가능한 코딩을 한다.
6. 가급적 !important로 기존 코드를 덮어쓰려 시도하지 말고 정리하며 코딩한다.
